## 8.2 完整代码示例

```python
"""
连扳接力量化策略 - 完整版
作者: 量化学习者
功能: 昨天涨停 + 今天换手 → 次日卖出
"""

import akshare as ak
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
from tqdm import tqdm


# ==================== 数据层 ====================

def get_stock_data(stock_code, days=30):
    """获取股票数据"""

    end_date = datetime.now().strftime('%Y%m%d')
    start_date = (datetime.now() - timedelta(days=days*2)).strftime('%Y%m%d')

    try:
        df = ak.stock_zh_a_hist(
            symbol=stock_code,
            period='daily',
            start_date=start_date,
            end_date=end_date
        )

        # 数据清洗
        df = df.sort_values('日期')
        df = df[['日期', '开盘', '收盘', '最高', '最低', '成交量', '涨跌幅']]
        df.columns = ['date', 'open', 'close', 'high', 'low', 'volume', 'pct_change']
        df['date'] = pd.to_datetime(df['date'])
        df = df.set_index('date')

        # 计算指标
        df['ma5'] = df['close'].rolling(window=5).mean()
        df['ma20'] = df['close'].rolling(window=20).mean()
        df['vol_ma5'] = df['volume'].rolling(window=5).mean()
        df['volume_ratio'] = df['volume'] / df['vol_ma5']

        return df
    except:
        return None


def get_all_stocks():
    """获取全市场股票列表"""
    try:
        df = ak.stock_zh_a_stock_symbols()
        # 过滤：只保留主板和创业板（排除ST、北交所等）
        df = df[df['代码'].str.startswith(('000', '001', '002', '300', '600', '601', '603'))]
        return df
    except:
        return None


# ==================== 策略层 ====================

class LianBanStrategy:
    """连扳接力策略"""

    def __init__(self):
        # 策略参数
        self.limit_pct = 9.9         # 涨停幅度
        self.vol_ratio = 1.5          # 成交量放大倍数
        self.ma_period = 20           # 均线周期
        self.hold_days = 1            # 持有天数
        self.stop_loss = -5           # 止损线 (%)

        # 交易记录
        self.trades = []

    def check_buy_signal(self, df):
        """检查买入信号"""

        if df is None or len(df) < self.ma_period + 2:
            return False

        yesterday = df.iloc[-2]
        today = df.iloc[-1]

        # 条件1：昨天涨停
        cond1 = yesterday['pct_change'] >= self.limit_pct

        # 条件2：今天有换手（非一字板）
        cond2 = today['open'] < today['close']

        # 条件3：成交量放大
        cond3 = today['volume_ratio'] > self.vol_ratio

        # 条件4：站上MA20（趋势向上）
        cond4 = today['close'] > today['ma20']

        return cond1 and cond2 and cond3 and cond4

    def check_sell_signal(self, df, hold_days):
        """检查卖出信号"""

        if df is None or len(df) < 2:
            return False

        today = df.iloc[-1]

        # 卖出条件1：持有超过N天
        if hold_days >= self.hold_days:
            return True

        # 卖出条件2：止损
        if today['pct_change'] <= self.stop_loss:
            return True

        return False

    def run(self, stock_code, start_date, end_date):
        """运行策略回测"""

        # 获取数据
        df = get_stock_data(stock_code, days=365)
        if df is None or len(df) < self.ma_period + 5:
            return None

        # 过滤日期范围
        df = df[start_date:end_date]

        # 模拟交易
        position = None
        hold_days = 0
        trades = []

        for i in range(self.ma_period + 2, len(df)):
            current_df = df.iloc[:i+1]

            # 有持仓
            if position:
                hold_days += 1

                # 检查卖出信号
                if self.check_sell_signal(current_df, hold_days):
                    sell_price = df.iloc[i]['open']
                    profit_pct = (sell_price - position['buy_price']) / position['buy_price'] * 100

                    trades.append({
                        'stock': stock_code,
                        'buy_date': position['buy_date'],
                        'buy_price': position['buy_price'],
                        'sell_date': df.index[i].date(),
                        'sell_price': sell_price,
                        'hold_days': hold_days,
                        'profit_pct': profit_pct
                    })

                    position = None
                    hold_days = 0

            # 无持仓，检查买入信号
            else:
                if self.check_buy_signal(current_df):
                    buy_price = df.iloc[i]['open']
                    position = {
                        'buy_date': df.index[i].date(),
                        'buy_price': buy_price
                    }
                    hold_days = 0

        return trades


# ==================== 回测分析 ====================

def backtest_analysis(stocks, start_date, end_date):
    """回测分析"""

    strategy = LianBanStrategy()
    all_trades = []

    print(f"开始回测 {len(stocks)} 只股票...")

    for code in tqdm(stocks):
        trades = strategy.run(code, start_date, end_date)
        if trades:
            all_trades.extend(trades)

    if not all_trades:
        print("没有符合条件的交易")
        return None

    # 统计结果
    df_trades = pd.DataFrame(all_trades)

    wins = df_trades[df_trades['profit_pct'] > 0]
    losses = df_trades[df_trades['profit_pct'] <= 0]

    result = {
        'total_trades': len(df_trades),
        'win_trades': len(wins),
        'loss_trades': len(losses),
        'win_rate': len(wins) / len(df_trades) * 100,
        'avg_profit': df_trades['profit_pct'].mean(),
        'total_profit': df_trades['profit_pct'].sum(),
        'avg_win': wins['profit_pct'].mean() if len(wins) > 0 else 0,
        'avg_loss': losses['profit_pct'].mean() if len(losses) > 0 else 0,
        'max_profit': df_trades['profit_pct'].max(),
        'max_loss': df_trades['profit_pct'].min(),
        'trades': df_trades
    }

    return result


# ==================== 主程序 ====================

if __name__ == '__main__':
    # 1. 获取股票列表（测试用前100只）
    stocks_df = get_all_stocks()
    test_stocks = stocks_df['代码'].head(100).tolist()

    # 2. 运行回测
    start_date = '20230101'
    end_date = '20231231'

    result = backtest_analysis(test_stocks, start_date, end_date)

    # 3. 输出结果
    if result:
        print("\n" + "="*50)
        print("回测结果统计")
        print("="*50)
        print(f"交易次数: {result['total_trades']}")
        print(f"盈利次数: {result['win_trades']}")
        print(f"亏损次数: {result['loss_trades']}")
        print(f"胜率: {result['win_rate']:.1f}%")
        print(f"平均收益: {result['avg_profit']:.2f}%")
        print(f"总收益: {result['total_profit']:.2f}%")
        print(f"平均盈利: {result['avg_win']:.2f}%")
        print(f"平均亏损: {result['avg_loss']:.2f}%")
        print(f"最大单笔盈利: {result['max_profit']:.2f}%")
        print(f"最大单笔亏损: {result['max_loss']:.2f}%")
```

## 8.4 策略优化方向

| 方向 | 说明 | 效果 |
|-----|------|------|
| **添加板块过滤** | 只做热门板块 | 提高胜率 |
