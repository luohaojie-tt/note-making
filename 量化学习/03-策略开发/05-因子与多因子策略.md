## 一、因子分类详解

### 1.1 价格/技术因子

## 二、因子计算示例

```python
import akshare as ak
import pandas as pd
import numpy as np

def calculate_factors(df):
    """
    计算所有因子
    df: 包含 date, open, close, high, low, volume 数据
    """

    # ============== 价格因子 ==============

    # 日收益率
    df['return'] = df['close'].pct_change()

    # 5日、20日收益率
    df['return_5d'] = df['close'].pct_change(5)
    df['return_20d'] = df['close'].pct_change(20)

    # 均线
    for window in [5, 10, 20, 60]:
        df[f'ma{window}'] = df['close'].rolling(window).mean()

    # 均线偏离度
    df['ma20_deviation'] = (df['close'] - df['ma20']) / df['ma20'] * 100

    # ============== 成交量因子 ==============

    # 换手率（需要流通股本，这里用成交量近似）
    df['vol_ma5'] = df['volume'].rolling(5).mean()
    df['volume_ratio'] = df['volume'] / df['vol_ma5']

    # ============== 波动率因子 ==============

    # 20日波动率
    df['volatility_20d'] = df['return'].rolling(20).std()

    # ============== 动量因子 ==============

    # 动量：过去N天累计收益
    df['momentum_5d'] = df['close'].pct_change(5)
    df['momentum_20d'] = df['close'].pct_change(20)

    # ============== 涨停相关因子 ==============

    # 是否涨停
    df['is_limit_up'] = (df['pct_change'] >= 9.9).astype(int)

    # 连板计数
    df['limit_streak'] = df['is_limit_up'].groupby(
        (df['is_limit_up'] != df['is_limit_up'].shift()).cumsum()
    ).cumsum() * df['is_limit_up']

    return df
```

## 三、因子预处理

### 3.1 缺失值处理

```python
def handle_missing(df):
    """处理缺失值"""

    # 方法1：删除有缺失的行
    df = df.dropna()

    # 方法2：用均值填充
    df = df.fillna(df.mean())

    # 方法3：用前向填充（时间序列常用）
    df = df.fillna(method='ffill')

    return df
```

### 3.2 异常值处理（去极值）

```python
def winsorize(df, column, n_std=3):
    """去极值：把超过n倍标准差的值替换为边界值"""

    mean = df[column].mean()
    std = df[column].std()

    lower = mean - n_std * std
    upper = mean + n_std * std

    df[column] = df[column].clip(lower, upper)

    return df
```

### 3.3 标准化

```python
from sklearn.preprocessing import StandardScaler

def standardize(df):
    """标准化：使因子均值为0，标准差为1"""

    scaler = StandardScaler()

    # 只标准化数值列
    numeric_cols = df.select_dtypes(include=[np.number]).columns

    df[numeric_cols] = scaler.fit_transform(df[numeric_cols])

    return df
```

### 3.4 市值中性化

```python
def neutralize(df, factor_col, market_cap_col='market_cap'):
    """市值中性化：剥离市值对因子的影响"""

    # 用残差法：因子 = a * 市值 + 残差
    # 残差就是剥离市值后的纯因子效应

    from sklearn.linear_model import LinearRegression

    X = df[[market_cap_col]].values
    y = df[factor_col].values

    model = LinearRegression()
    model.fit(X, y)

    # 残差
    df[f'{factor_col}_neutral'] = y - model.predict(X)

    return df
```

## 四、多因子策略构建

### 4.1 因子筛选流程

```
Step 1: 候选因子池
    ↓
Step 2: 单因子测试 (IC分析)
    ↓
Step 3: 相关性分析 (去除冗余)
    ↓
Step 4: 多因子组合
    ↓
Step 5: 回测验证
```

### 4.2 单因子测试

```python
def single_factor_test(factor_data, return_data):
    """
    单因子有效性测试
    factor_data: 因子值序列
    return_data: 下期收益序列
    """

    # 计算IC (信息系数)
    ic = factor_data.corr(return_data)

    # 计算IR (信息比率)
    ic_series = factor_data.rolling(60).corr(return_data)  # 滚动60天IC
    ir = ic_series.mean() / ic_series.std()

    return {
        'IC': ic,
        'IR': ir,
        'IC_series': ic_series
    }
```

### 4.3 因子相关性分析

```python
def factor_correlation(factors_df):
    """计算因子相关性矩阵"""

    corr_matrix = factors_df.corr()

    # 找出高度相关的因子对
    high_corr_pairs = []
    for i in range(len(corr_matrix.columns)):
        for j in range(i+1, len(corr_matrix.columns)):
            if abs(corr_matrix.iloc[i, j]) > 0.7:
                high_corr_pairs.append((
                    corr_matrix.columns[i],
                    corr_matrix.columns[j],
                    corr_matrix.iloc[i, j]
                ))

    return corr_matrix, high_corr_pairs
```

### 4.4 多因子合成方法

#### 方法1：等权相加

```python
def equal_weight(factors_df):
    """等权相加"""

    # 标准化后的因子等权相加
    combined = factors_df.mean(axis=1)

    return combined
```

#### 方法2：IC加权

```python
def ic_weighted(factors_df, ic_values):
    """IC加权：IC越高的因子权重越大"""

    weights = ic_values / ic_values.sum()
    combined = (factors_df * weights).sum(axis=1)

    return combined
```

#### 方法3：回归法

```python
def regression_weight(factors_df, returns):
    """用回归确定因子权重"""

    from sklearn.linear_model import LinearRegression

    model = LinearRegression()
    model.fit(factors_df, returns)

    # 权重就是回归系数
    weights = pd.Series(model.coef_, index=factors_df.columns)

    # 预测值
    combined = model.predict(factors_df)

    return combined, weights
```

### 4.5 完整多因子策略示例

```python
class MultiFactorStrategy:
    """多因子选股策略"""

    def __init__(self):
        # 选取的因子
        self.factors = [
            'return_20d',      # 20日动量
            'volume_ratio',    # 成交量放大
            'ma20_deviation',  # 均线偏离
            'limit_streak',    # 连板天数
            'volatility_20d'   # 波动率
        ]

        # 因子权重（可优化）
        self.weights = [0.2, 0.2, 0.2, 0.3, 0.1]

    def calculate_score(self, df):
        """计算综合得分"""

        # 标准化
        for factor in self.factors:
            if factor in df.columns:
                df[f'{factor}_std'] = (df[factor] - df[factor].mean()) / df[factor].std()

        # 计算加权得分
        std_cols = [f'{f}_std' for f in self.factors if f'{f}_std' in df.columns]
        df['score'] = 0
        for i, col in enumerate(std_cols):
            df['score'] += df[col] * self.weights[i]

        return df

    def select_stocks(self, df, top_n=10):
        """选top N只股票"""

        df = self.calculate_score(df)
        selected = df.nlargest(top_n, 'score')

        return selected
```

## 七、IC值（信息系数）详解

### 7.1 什么是IC值？

> **IC = 因子值 与 下期收益 的相关性**

```
IC = corr(因子值, 下期收益)
```

- **皮尔逊相关系数**，范围是 -1 到 1
- **含义**：因子大的股票，下期收益是否也高？

### 7.2 举例说明

假设有5只股票的换手率因子和下期收益：

| 股票 | 换手率（因子值） | 下期收益 |
|-----|----------------|---------|
| A | 80% | +5% |
| B | 60% | +3% |
| C | 40% | +1% |
| D | 20% | -2% |
| E | 10% | -4% |

**结论**：因子值越高，收益越高 → IC > 0（正相关）

### 7.3 IC值计算代码

```python
import numpy as np
import pandas as pd

def calculate_ic(factor_values, returns):
    """
    计算IC值
    factor_values: 因子值（比如换手率、成交量等）
    returns: 对应的下期收益
    """

    # 方法1：numpy计算
    n = len(factor_values)
    mean_factor = np.mean(factor_values)
    mean_return = np.mean(returns)

    # 分子：协方差
    covariance = np.sum((factor_values - mean_factor) * (returns - mean_return)) / n

    # 分母：两个标准差的乘积
    std_factor = np.std(factor_values)
    std_return = np.std(returns)

    # IC = 协方差 / (标准差1 * 标准差2)
    ic = covariance / (std_factor * std_return)

    return ic


def calculate_ic_pandas(factor_df):
    """
    用pandas计算IC（更简洁）
    """

    # IC = 因子与收益的相关系数
    ic = factor_df['factor'].corr(factor_df['return'])

    return ic
```

### 7.4 滚动IC计算

实际应用中，我们计算一段时间内的滚动IC：

```python
def rolling_ic(factor_series, return_series, window=60):
    """
    滚动IC：计算一段时间内的IC序列
    window: 滚动天数（比如60天）
    """

    # 合并数据
    df = pd.DataFrame({
        'factor': factor_series,
        'return': return_series
    }).dropna()

    # 滚动计算IC
    ic_series = df['factor'].rolling(window).corr(df['return'])

    # 统计
    mean_ic = ic_series.mean()      # 平均IC
    std_ic = ic_series.std()        # IC标准差
    ir = mean_ic / std_ic           # IR = IC均值 / IC标准差

    return {
        'IC序列': ic_series,
        '平均IC': mean_ic,
        'IC标准差': std_ic,
        'IR': ir
    }
```

### 7.5 IC值判断标准

| IC值 | 因子效果 |
|-----|---------|
| > 0.05 | 很好！因子预测能力强 |
| 0.02 ~ 0.05 | 一般，可用 |
| 0 ~ 0.02 | 较弱 |
| < 0 | 反向！因子大反而收益低 |

### 7.6 IR（信息比率）

```
IR = 平均IC / IC标准差
```

- **IR > 0.5**：因子稳定有效
- **IR < 0.3**：因子时好时坏，不稳定

**简单理解**：
- IC  = 因子"准不准"
- IR  = 因子"稳不稳"

## 本课重点

1. **因子三大类**：价格/技术因子、基本面因子、资金流向因子
2. **因子处理流程**：缺失值 → 异常值 → 标准化 → 中性化
3. **多因子合成**：等权、IC加权、回归法
4. **IC值**：因子预测能力，IC > 0.02 算有效
5. **IR值**：因子稳定性，IR > 0.5 算稳定
6. **你的连扳因子**：连板天数 + 成交量 + 换手率 + 封板率
